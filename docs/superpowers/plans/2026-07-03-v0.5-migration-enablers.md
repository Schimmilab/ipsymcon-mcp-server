def _collect_custom_profiles(node: dict[str, Any], names: set[str]) -> None:
    """Gather every non-system profile name (no '~' prefix) referenced in the subtree."""
    profile = node.get("profile")
    if profile and not str(profile).startswith("~"):
        names.add(str(profile))
    for child in node.get("children") or []:
        _collect_custom_profiles(child, names)

```python
async def _import_profiles(client: IPSClient, profiles: dict[str, Any]) -> dict[str, list[str]]:
    """Create every exported custom profile that is missing on the target.

    Existing profiles are left untouched (recorded under 'existed'); definitions the export
    could not read (or non-dicts) land under 'failed'.
    """
    created: list[str] = []
    existed: list[str] = []
    failed: list[str] = []
    for name, d in profiles.items():
        if not isinstance(d, dict) or "error" in d:
            failed.append(name)
            continue
        if await client.call("IPS_VariableProfileExists", [name]):
            existed.append(name)
            continue
        ptype = int(d.get("ProfileType", 0))
        await client.call("IPS_CreateVariableProfile", [name, ptype])
        await client.call("IPS_SetVariableProfileText", [name, d.get("Prefix") or "", d.get("Suffix") or ""])
        if d.get("Icon"):
            await client.call("IPS_SetVariableProfileIcon", [name, d["Icon"]])
        if ptype in (1, 2):  # integer/float carry min/max/step
            await client.call("IPS_SetVariableProfileValues",
                              [name, d.get("MinValue") or 0, d.get("MaxValue") or 0, d.get("StepSize") or 0])
        if ptype == 2:  # float carries digits
            await client.call("IPS_SetVariableProfileDigits", [name, int(d.get("Digits") or 0)])
        for assoc in d.get("Associations") or []:
            await client.call("IPS_SetVariableProfileAssociation",
                              [name, assoc.get("Value"), assoc.get("Name") or "", assoc.get("Icon") or "",
                               assoc.get("Color") if assoc.get("Color") is not None else -1])
        created.append(name)
    return {"created": created, "existed": existed, "failed": failed}
```

def _index_children(node: dict[str, Any]) -> dict[tuple[Any, Any], dict[str, Any]]:
    return {(c.get("name"), c.get("type")): c for c in node.get("children") or []}


def _diff_nodes(
    src: dict[str, Any], tgt: dict[str, Any], path: str,
    report: dict[str, list], compare_values: bool,
) -> None:
    """Compare two exported nodes recursively; children are matched by (name, type).

    Siblings sharing (name, type) collapse onto one entry — rename duplicates on the
    source for a reliable comparison.
    """
    if src.get("type") == "Variable":
        fields = ("variable_type", "profile", "value") if compare_values else ("variable_type", "profile")
        for field in fields:
            if src.get(field) != tgt.get(field):
                report["variable_mismatches"].append({
                    "path": path, "field": field, "source": src.get(field), "target": tgt.get(field),
                    "source_id": src.get("id"), "target_id": tgt.get("id")})
    if src.get("type") == "Script" and src.get("content") != tgt.get("content"):
        report["script_content_mismatches"].append(
            {"path": path, "source_id": src.get("id"), "target_id": tgt.get("id")})
    src_children = _index_children(src)
    tgt_children = _index_children(tgt)
    for key, child in src_children.items():
        child_path = f"{path}/{key[0]}"
        if key not in tgt_children:
            report["missing_on_target"].append(
                {"path": child_path, "type": key[1], "source_id": child.get("id")})
        else:
            _diff_nodes(child, tgt_children[key], child_path, report, compare_values)
    for key, child in tgt_children.items():
        if key not in src_children:
            report["extra_on_target"].append(
                {"path": f"{path}/{key[0]}", "type": key[1], "target_id": child.get("id")})


@mcp.tool(
    name="ips_compare_subtrees",
    annotations={"title": "Compare subtrees across instances", "readOnlyHint": True,
                 "destructiveHint": False, "idempotentHint": True, "openWorldHint": True},
)
async def ips_compare_subtrees(params: CompareSubtreesInput) -> str:
    """Compare a subtree on a SOURCE instance against one on a TARGET instance (read-only).

    The migration verification tool: exports both subtrees (like ips_export_subtree, without
    profiles/archive detail), matches children by (name, type) path and reports
    missing_on_target / extra_on_target / variable_mismatches (type, profile, value) /
    script_content_mismatches, plus an overall in_sync flag. Events, instances and links are
    compared by PRESENCE only — their configs contain instance-specific IDs by design.
    Set compare_values=False for a structure-only check while live values drift.
    Returns the report JSON.
    """
    try:
        src = await _export_node(_client(params.source_instance), params.source_root_id, 0, params.max_depth)
        tgt = await _export_node(_client(params.target_instance), params.target_root_id, 0, params.max_depth)
        report: dict[str, list] = {"missing_on_target": [], "extra_on_target": [],
                                   "variable_mismatches": [], "script_content_mismatches": []}
        _diff_nodes(src, tgt, str(src.get("name") or "root"), report, params.compare_values)
        in_sync = not any(report.values())
        return _dumps({
            "source": {"instance": params.source_instance, "root_id": params.source_root_id},
            "target": {"instance": params.target_instance, "root_id": params.target_root_id},
            "in_sync": in_sync,
            "counts": {k: len(v) for k, v in report.items()},
            **report,
        })
    except Exception as e:  # noqa: BLE001
        return _handle_error(e)
